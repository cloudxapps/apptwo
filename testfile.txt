/**
 * Genesys Cloud Function – Trunk Monitoring (SIP failure check from SIP traces)
 *
 * Invoked from an Architect workflow (via a Function Data Action) after a call ends.
 * 1. Gets an OAuth token (client credentials)
 * Input: conversationId (required), ani and dnis (optional - if given, only failures on
 *        the SIP leg from this ANI (fromUser) / to this DNIS (toUser) are reported;
 *        avoids duplicate alerts)
 * 2. Searches SIP traces (GET /api/v2/telephony/siptraces) by conversationId
 *    over the last 60 minutes
 * 3. Flags SIP failure responses listed in TARGET_SIP_CODES (default: any 4xx/5xx/6xx),
 *    except IGNORE_SIP_CODES (default: 401/407, normal authentication challenges)
 * 4. Returns to Architect: connid, method, ani, dnis, traceFound, message
 *    (method = failure code(s) found, e.g. "487" or "480,500"; empty = no failure)
 *    Alerting (e.g. email to a DL) is done in the Architect workflow.
 *
 * In SIP traces, responses carry the status code in "method" (e.g. "487")
 * and the request they answer in "cseq" (e.g. "1 INVITE").
 *
 * Runtime: Node.js 18+ / 20 (uses global fetch, no dependencies)
 * Handler: index.handler
 *
 * Credentials are hard-coded below (GC_CLIENT_ID, GC_CLIENT_SECRET, GC_REGION).
 */

'use strict';

// ---- Genesys Cloud OAuth (Client Credentials) - fill these in ----
const GC_CLIENT_ID = 'PASTE_CLIENT_ID_HERE';
const GC_CLIENT_SECRET = 'PASTE_CLIENT_SECRET_HERE';
const GC_REGION = 'mypurecloud.com'; // e.g. usw2.pure.cloud, mypurecloud.ie, aps1.pure.cloud
// -------------------------------------------------------------------

const TRACE_WINDOW_MIN = 60; // how far back to search SIP traces (minutes)

// SIP codes treated as failures. Use exact codes ('487') or classes ('5xx' = 500-599).
// Examples: ['487', '480', '500']  or  ['480', '487', '5xx']
const TARGET_SIP_CODES = ['4xx', '5xx', '6xx'];

// SIP codes never treated as failures (401/407 are normal authentication challenges)
const IGNORE_SIP_CODES = ['401', '407'];

function isFailureCode(code) {
  if (!/^\d{3}$/.test(code) || IGNORE_SIP_CODES.includes(code)) return false;
  return TARGET_SIP_CODES.some((p) => {
    const pat = String(p).toLowerCase();
    return /^\dxx$/.test(pat) ? code[0] === pat[0] : code === pat;
  });
}

function getConfig() {
  return {
    clientId: GC_CLIENT_ID,
    clientSecret: GC_CLIENT_SECRET,
    region: GC_REGION,
  };
}

async function getToken(cfg) {
  const basic = Buffer.from(`${cfg.clientId}:${cfg.clientSecret}`).toString('base64');
  const res = await fetch(`https://login.${cfg.region}/oauth/token`, {
    method: 'POST',
    headers: {
      Authorization: `Basic ${basic}`,
      'Content-Type': 'application/x-www-form-urlencoded',
    },
    body: 'grant_type=client_credentials',
  });
  if (!res.ok) throw new Error(`OAuth failed: HTTP ${res.status}`);
  return (await res.json()).access_token;
}

async function apiGet(cfg, token, path) {
  const res = await fetch(`https://api.${cfg.region}${path}`, {
    headers: { Authorization: `Bearer ${token}` },
  });
  if (res.status === 404) return null;
  if (!res.ok) throw new Error(`GET ${path.split('?')[0]} failed: HTTP ${res.status}`);
  return res.json();
}

const stripTel = (v) => String(v || '').replace(/^tel:|^sip:/i, '').split('@')[0].split(';')[0];

// Compare phone numbers on their last 10 digits ("tel:+13135551234" == "3135551234")
const last10 = (v) => stripTel(v).replace(/\D/g, '').slice(-10);
const sameNumber = (a, b) => last10(a) !== '' && last10(a) === last10(b);

async function searchSipTraces(cfg, token, conversationId) {
  const to = new Date();
  const from = new Date(to.getTime() - TRACE_WINDOW_MIN * 60 * 1000);
  const path = `/api/v2/telephony/siptraces?dateStart=${encodeURIComponent(from.toISOString())}`
    + `&dateEnd=${encodeURIComponent(to.toISOString())}`
    + `&conversationId=${encodeURIComponent(conversationId)}`;

  const body = await apiGet(cfg, token, path);
  const rows = Array.isArray(body) ? body : (body && (body.data || body.entities)) || [];
  return rows;
}

/** Find SIP failure responses (e.g. 487, 480, 500) in the trace rows. */
function analyzeTraces(rows, ani, dnis) {
  const r = {
    matched: false,
    matchedCodes: new Set(),
    hasCancel: false,
    fromUser: '',
    toUser: '',
    firstFrom: '',
    firstTo: '',
  };
  for (const m of rows) {
    const method = String(m.method || '').trim();
    r.firstFrom = r.firstFrom || m.fromUser || '';
    r.firstTo = r.firstTo || m.toUser || '';
    if (method.toUpperCase() === 'CANCEL') r.hasCancel = true;
    // If ANI / DNIS were passed in, only count failures on the SIP leg from that ANI / to that DNIS
    const aniMatches = !ani || sameNumber(ani, m.fromUser);
    const dnisMatches = !dnis || sameNumber(dnis, m.toUser);
    if (isFailureCode(method) && aniMatches && dnisMatches) {
      r.matched = true;
      r.matchedCodes.add(method);
      r.fromUser = r.fromUser || m.fromUser || '';
      r.toUser = r.toUser || m.toUser || '';
    }
  }
  return r;
}

const SIP_TEXT = {
  400: 'Bad Request', 403: 'Forbidden', 404: 'Not Found', 408: 'Request Timeout',
  480: 'Temporarily Unavailable', 481: 'Call Does Not Exist', 484: 'Address Incomplete',
  486: 'Busy Here', 487: 'Request Terminated', 488: 'Not Acceptable Here',
  500: 'Server Internal Error', 502: 'Bad Gateway', 503: 'Service Unavailable',
  504: 'Server Time-out', 603: 'Decline',
};
const describe = (c) => (SIP_TEXT[c] ? `${c} ${SIP_TEXT[c]}` : String(c));

/** Runs the SIP trace check for one conversation. */
async function checkConversation(event, context) {
  const conversationId = event && event.conversationId;
  const out = {
    conversationId: conversationId || '',
    traceFound: false,
    matchedCodes: '',
    hasCancel: false,
    ani: '',
    dnis: '',
    error: '',
  };

  try {
    if (!conversationId) throw new Error('conversationId is required');
    const cfg = getConfig();
    if (!cfg.clientId || cfg.clientId.startsWith('PASTE_') || !cfg.clientSecret || cfg.clientSecret.startsWith('PASTE_')) {
      throw new Error('GC_CLIENT_ID / GC_CLIENT_SECRET not set in index.js');
    }

    const token = await getToken(cfg);
    const rows = await searchSipTraces(cfg, token, conversationId);
    out.traceFound = rows.length > 0;

    const t = analyzeTraces(rows, event.ani, event.dnis);
    Object.assign(out, {
      matchedCodes: [...t.matchedCodes].join(','),
      hasCancel: t.hasCancel,
      ani: t.fromUser || t.firstFrom,
      dnis: t.toUser || t.firstTo,
    });
  } catch (err) {
    out.error = String(err && err.message ? err.message : err);
  }
  return out;
}

const cleanNum = (v) => stripTel(v);

/** Human-readable result, e.g. "SIP failure: 487 Request Terminated" */
function buildMessage(r) {
  if (r.error) return `Check failed: ${r.error}`;
  if (!r.traceFound) return 'No SIP trace found for this call';
  if (!r.matchedCodes) return 'No failure';
  const codes = r.matchedCodes.split(',').map(describe).join(', ');
  return `SIP failure: ${codes}${r.hasCancel ? ' (CANCEL seen)' : ''}`;
}

/** Function entry point: returns connid, method, ani, dnis, traceFound, message. */
exports.handler = async (event, context) => {
  const r = await checkConversation(event, context);
  if (r.error) console.error(`Trunk Monitoring error for ${r.conversationId}: ${r.error}`);
  return {
    connid: r.conversationId,
    method: r.matchedCodes,
    ani: cleanNum(r.ani),
    dnis: cleanNum(r.dnis),
    traceFound: r.traceFound,
    message: buildMessage(r),
  };
};

==============================================================================
 5. SETUP STEPS
==============================================================================

A. OAUTH CLIENT
   Admin > Integrations > OAuth > Add client
   - Grant type: Client Credentials
   - Role: permission to view SIP traces (same as your user uses for Call History)
   - Divisions: All divisions (avoids HTTP 403 on calls in other divisions)
   - Copy the Client ID and Secret into index.js (GC_CLIENT_ID, GC_CLIENT_SECRET)
   - Set GC_REGION to your org region (e.g. mypurecloud.com, usw2.pure.cloud)

B. BUILD THE UPLOAD ZIP  (PowerShell, in the folder with index.js + package.json)
   Compress-Archive -Path index.js,package.json -DestinationPath trunk-monitoring-function.zip -Force

C. FUNCTION DATA ACTION
   Admin > Integrations > Actions > your Function action
   - Contracts tab: paste the input and output contracts above (JSON view)
   - Function tab: upload trunk-monitoring-function.zip
       Handler: index.handler    Runtime: Node.js 20    Timeout: 15-30 s
   - Test tab: enter conversationId (+ ani, dnis optional) > Run
       (for calls older than 60 min, temporarily set TRACE_WINDOW_MIN = 2880,
        and set it back to 60 before publishing)
   - Save & Publish

D. ARCHITECT WORKFLOW
   Input variables (String, "Input to flow" checked):
     Flow.conversationId, Flow.ani, Flow.dnis
   Steps:
     1. Call Data Action > Trunk Monitoring action
          conversationId = Flow.conversationId
          ani            = Flow.ani
          dnis           = Flow.dnis
        Save outputs to: Flow.ConID (connid), Flow.method, Flow.ANI (ani),
                         Flow.dnis (dnis), Flow.traceFound, Flow.message
     2. Decision: Flow.method != ""
          Yes > send email to DL
          No  > End Workflow
   Save & Publish.

   Email subject expression:
     "Trunk Monitoring: " + Flow.message + " | ANI: " + Flow.ANI + " | DNIS: " + Flow.dnis

   Email body expression:
     Append(
       "Found ", If(IsNotSetOrEmpty(Flow.message), "", Flow.message),
       " on ", If(IsNotSetOrEmpty(Flow.dnis), "", Flow.dnis),
       ". Here are the details: ",
       "ConnID: ", If(IsNotSetOrEmpty(Flow.ConID), "", Flow.ConID),
       " | DNIS: ", If(IsNotSetOrEmpty(Flow.dnis), "", Flow.dnis),
       " | ANI: ", If(IsNotSetOrEmpty(Flow.ANI), "", Flow.ANI)
     )
   Notes: text in Architect expressions uses double quotes (" "), not single (' ').
          ToDisplayString is not allowed in Workflow flows for strings.

E. TRIGGER
   Admin > Architect > Triggers > Add
     Name:          Trunk Monitoring
     Topic:         v2.detail.events.conversation.{id}.customer.end
     Target:        your Trunk Monitoring workflow
     Data format:   TopLevelPrimitives
     Delay:         120 seconds
     Event TTL:     3600 seconds
     Match criteria: mediaType Equal VOICE  (optionally direction / dnis)
   The event's conversationId, ani and dnis fill Flow.conversationId,
   Flow.ani and Flow.dnis automatically.

F. HOW IT DECIDES
   - Searches SIP traces for the conversationId over the last 60 minutes.
   - A failure = any code in TARGET_SIP_CODES (default 4xx/5xx/6xx),
     except IGNORE_SIP_CODES (default 401, 407).
   - If ani / dnis are passed, a failure only counts when that SIP message is
     FROM the same ANI (fromUser) and TO the same DNIS (toUser). This stops
     duplicate emails when one call triggers several runs.
   - Output: method = failure code(s) or empty; message = readable result;
     traceFound = true/false.
   - If two runs still pass the same ANI and DNIS, both will alert; that case
     needs a data table to remember already-reported failures.

G. SECURITY
   The client secret is in plain text in index.js and the zip. Limit the OAuth
   role to SIP trace view only and do not share or commit the file.
