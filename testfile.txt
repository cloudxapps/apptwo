---------------------------------------------------------------------
 Q3. Segment rows vs distinct calls (yesterday)
     Purpose: show why we count conversations, not rows.
---------------------------------------------------------------------

SELECT COUNT(*)                        AS segment_rows,
       COUNT(DISTINCT conversation_id) AS distinct_calls
FROM A3S_GCD.PUBLIC.CONVERSATIONS_SEGMENT_FACT
WHERE CONVERT_TIMEZONE('UTC', 'America/New_York',
        TO_TIMESTAMP_NTZ(segment_start_ts))::date
      = DATEADD(day, -1, CONVERT_TIMEZONE('America/New_York', CURRENT_TIMESTAMP())::date);


---------------------------------------------------------------------
 Q4. Channels and directions (yesterday)
     Purpose: confirm the voice / e-mail values and inbound label.
     NOTE: if it errors, replace media_type / direction with the
     column names from Q1.
---------------------------------------------------------------------

SELECT media_type,
       direction,
       COUNT(DISTINCT conversation_id) AS calls
FROM A3S_GCD.PUBLIC.CONVERSATIONS_SEGMENT_FACT
WHERE CONVERT_TIMEZONE('UTC', 'America/New_York',
        TO_TIMESTAMP_NTZ(segment_start_ts))::date
      = DATEADD(day, -1, CONVERT_TIMEZONE('America/New_York', CURRENT_TIMESTAMP())::date)
GROUP BY 1, 2
ORDER BY calls DESC;


---------------------------------------------------------------------
 Q5. PCI Pal - do both legs share the same session ID? (yesterday)
     Purpose: decide how to pair and drop the duplicate leg.
     conversations = 2  -> both legs share the session ID
     conversations = 1  -> second leg does not carry it
---------------------------------------------------------------------

SELECT value                                    AS pcipal_session,
       COUNT(DISTINCT conversation_id)          AS conversations,
       LISTAGG(DISTINCT conversation_id, ', ')  AS conversation_ids
FROM attributes
WHERE keyname ILIKE '%PCIPalSessionId%'
  AND CONVERT_TIMEZONE('UTC', 'America/New_York',
        TO_TIMESTAMP_NTZ(start_date_time_key))::date
      = DATEADD(day, -1, CONVERT_TIMEZONE('America/New_York', CURRENT_TIMESTAMP())::date)
GROUP BY 1
ORDER BY conversations DESC;


---------------------------------------------------------------------
 Q6. PCI Pal - summary of Q5 (yesterday)
     Purpose: how many sessions have 1 vs 2+ conversations.
---------------------------------------------------------------------

SELECT conversations,
       COUNT(*) AS sessions
FROM (
  SELECT value                           AS pcipal_session,
         COUNT(DISTINCT conversation_id) AS conversations
  FROM attributes
  WHERE keyname ILIKE '%PCIPalSessionId%'
    AND CONVERT_TIMEZONE('UTC', 'America/New_York',
          TO_TIMESTAMP_NTZ(start_date_time_key))::date
        = DATEADD(day, -1, CONVERT_TIMEZONE('America/New_York', CURRENT_TIMESTAMP())::date)
  GROUP BY 1
)
GROUP BY 1
ORDER BY 1;
